 _ThresholdLow("Low Threshold",Range(0,1))=0.25
 _ThresholdHigh("High Threshold",Range(0,1))=0.65
}

 Tags { "RenderType"="Opaque" }
 LOD 200
 CGPROGRAM
 #pragma surface surf CandleToon fullforwardshadows addshadow
 #pragma target 3.0

 struct Input { float3 worldPos; };
 half4 LightingCandleToon(SurfaceOutput s,half3 lightDir,half atten) {

  half band=ndl<_ThresholdLow?_ShadowLevel:(ndl<_ThresholdHigh?_MiddleLevel:1);
  return half4(s.Albedo*_LightColor0.rgb*band*atten,s.Alpha);
 }

 ENDCG
